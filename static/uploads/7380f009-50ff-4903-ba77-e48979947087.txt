    <script>
        const user = JSON.parse(localStorage.getItem("user"));
        if (!user || !user.id) {
            window.location.replace("/login");
        } else {
            document.getElementById("userCompanyName").innerText = user.company_name || user.email;
            loadStats();
            loadRequests();
            loadCarriersList();
        }

        function logout() {
            localStorage.removeItem("user");
            window.location.replace("/login");
        }

        function toggleCarrierSelection() {
            const sendToSelected = document.getElementById("sendToSelectedRadio").checked;
            const container = document.getElementById("carrierSelectionContainer");
            if (sendToSelected) {
                container.classList.remove("d-none");
            } else {
                container.classList.add("d-none");
            }
        }

        function selectAllCarriers(select) {
            const checkboxes = document.querySelectorAll(".carrier-item-check");
            checkboxes.forEach(cb => cb.checked = select);
        }

        async function loadCarriersList() {
            try {
                // user.id daxilindəki potensial səhv simvolları (məsələn '1:1') təmizləyib yalnız rəqəmi saxlayırıq
                const cleanId = typeof user.id === 'string' ? user.id.replace(/[^0-9]/g, '') : user.id;

                const res = await fetch(`/carriers/customer/${cleanId}`);
                const data = await res.json();
                const box = document.getElementById("carrierCheckboxList");
                
                let carriers = [];
                if (Array.isArray(data)) {
                    carriers = data;
                } else if (data.carriers && Array.isArray(data.carriers)) {
                    carriers = data.carriers;
                } else if (data.data && Array.isArray(data.data)) {
                    carriers = data.data;
                }

                if (res.ok && carriers.length > 0) {
                    let html = "";
                    for (let c of carriers) {
                        html += `
                            <div class="carrier-item-row" onclick="document.getElementById('carrier_${c.id}').click(); event.stopPropagation();">
                                <div class="form-check mb-0 w-100">
                                    <input class="form-check-input carrier-item-check me-2" type="checkbox" value="${c.id}" id="carrier_${c.id}" onclick="event.stopPropagation();">
                                    <label class="form-check-label fw-semibold text-dark" for="carrier_${c.id}" style="cursor: pointer;">
                                        ${c.company_name || c.name || 'Şirkət adı yoxdur'} <span class="text-muted fw-normal small">(${c.email || ''})</span>
                                    </label>
                                </div>
                            </div>
                        `;
                    }
                    box.innerHTML = html;
                } else {
                    box.innerHTML = `<span class="text-danger small p-2 d-block">Bazada daşıyıcı tapılmadı. Zəhmət olmasa yuxarıdan daşıyıcı əlavə edin.</span>`;
                }
            } catch (err) {
                console.error("Daşıyıcılar yüklənmədi:", err);
                document.getElementById("carrierCheckboxList").innerHTML = `<span class="text-danger small p-2 d-block">Daşıyıcıları yükləyərkən şəbəkə xətası baş verdi.</span>`;
            }
        }

        function toggleOptField(key) {
            const chk = document.getElementById(`chk_${key}`);
            if (key === 'info') {
                const container = document.getElementById(`inp_info_container`);
                if (chk.checked) {
                    container.classList.remove('d-none');
                } else {
                    container.classList.add('d-none');
                    document.getElementById(`inp_info_text`).value = '';
                    document.getElementById(`inp_info_file`).value = '';
                }
            } else {
                const input = document.getElementById(`inp_${key}`);
                if (chk.checked) {
                    input.classList.remove('d-none');
                } else {
                    input.classList.add('d-none');
                    input.value = '';
                }
            }
        }

        async function loadStats() {
            try {
                const cleanId = typeof user.id === 'string' ? user.id.replace(/[^0-9]/g, '') : user.id;
                const res = await fetch(`/customer/stats/${cleanId}`);
                const data = await res.json();
                if (res.ok) {
                    document.getElementById("statActiveRfqs").innerText = data.active_rfqs || 0;
                    document.getElementById("statIncomingQuotes").innerText = data.incoming_quotes || 0;
                    document.getElementById("statCompleted").innerText = data.completed_shipments || 0;
                    document.getElementById("statCarriersCount").innerText = data.carriers_count || 0;
                }
            } catch (err) {
                console.error("Statistika xətası:", err);
            }
        }

        async function uploadCarriers(e) {
            e.preventDefault();
            const fileInput = document.getElementById("carrierFile");
            const resultDiv = document.getElementById("uploadResult");

            if (!fileInput.files.length) {
                alert("Zəhmət olmasa Excel və ya CSV faylı seçin!");
                return;
            }

            const cleanId = typeof user.id === 'string' ? user.id.replace(/[^0-9]/g, '') : user.id;
            const formData = new FormData();
            formData.append("customer_id", cleanId);
            formData.append("file", fileInput.files[0]);

            resultDiv.innerHTML = `<span class="text-primary">Fayl yüklənir və emal olunur...</span>`;

            try {
                const res = await fetch("/carriers/upload-excel", {
                    method: "POST",
                    body: formData
                });
                const data = await res.json();
                if (res.ok) {
                    resultDiv.innerHTML = `<span class="text-success fw-semibold"><i class="bi bi-check-circle"></i> ${data.message}</span>`;
                    fileInput.value = "";
                    loadStats();
                    loadCarriersList();
                } else {
                    resultDiv.innerHTML = `<span class="text-danger">Xəta: ${data.detail}</span>`;
                }
            } catch (err) {
                resultDiv.innerHTML = `<span class="text-danger">Serverlə əlaqə qurulmadı!</span>`;
            }
        }

        async function createRfq(e) {
            e.preventDefault();

            const sendToAll = document.getElementById("sendToAllRadio").checked;
            let selectedCarriers = [];
            if (!sendToAll) {
                const checkboxes = document.querySelectorAll(".carrier-item-check:checked");
                checkboxes.forEach(cb => selectedCarriers.push(parseInt(cb.value)));
                if (selectedCarriers.length === 0) {
                    alert("Zəhmət olmasa siyahıdan ən azı bir daşıyıcı seçin və ya 'Bütün daşıyıcılara göndər' seçin!");
                    return;
                }
            }

            const requiredFieldsList = [
                `Truck Type: ${document.getElementById("truck_type").value}`
            ];

            if(document.getElementById("chk_volume").checked) {
                let val = document.getElementById("inp_volume").value.trim();
                if(val) requiredFieldsList.push(`Volume (CBM): ${val}`);
            }
            if(document.getElementById("chk_incoterm").checked) {
                let val = document.getElementById("inp_incoterm").value.trim();
                if(val) requiredFieldsList.push(`Incoterm: ${val}`);
            }
            if(document.getElementById("chk_adr").checked) {
                let val = document.getElementById("inp_adr").value.trim();
                if(val) requiredFieldsList.push(`Dangerous Goods (ADR): ${val}`);
            }
            if(document.getElementById("chk_temp").checked) {
                let val = document.getElementById("inp_temp").value.trim();
                if(val) requiredFieldsList.push(`Temperature Requirement: ${val}`);
            }
            if(document.getElementById("chk_deliv_date").checked) {
                let val = document.getElementById("inp_deliv_date").value.trim();
                if(val) requiredFieldsList.push(`Delivery Deadline: ${val}`);
            }
            if(document.getElementById("chk_attachment").checked) {
                let fileInput = document.getElementById("inp_attachment");
                if(fileInput.files.length > 0) {
                    requiredFieldsList.push(`Attachment: ${fileInput.files[0].name}`);
                }
            }
            if(document.getElementById("chk_info").checked) {
                let textVal = document.getElementById("inp_info_text").value.trim();
                let fileInput = document.getElementById("inp_info_file");
                let infoStr = textVal;
                if(fileInput.files.length > 0) {
                    infoStr += ` [Fayl: ${fileInput.files[0].name}]`;
                }
                if(infoStr) requiredFieldsList.push(`Additional Information: ${infoStr}`);
            }

            const cleanId = typeof user.id === 'string' ? user.id.replace(/[^0-9]/g, '') : user.id;
            const payload = {
                customer_id: parseInt(cleanId),
                origin: document.getElementById("origin").value,
                destination: document.getElementById("destination").value,
                cargo_type: document.getElementById("cargo_type").value,
                weight_kg: parseFloat(document.getElementById("weight_kg").value),
                volume_m3: parseFloat(document.getElementById("inp_volume")?.value) || 0,
                deadline: document.getElementById("deadline").value,
                required_fields: requiredFieldsList,
                send_to_all: sendToAll,
                carrier_ids: selectedCarriers
            };

            try {
                const res = await fetch("/requests/create", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(payload)
                });
                const data = await res.json();
                if (res.ok) {
                    alert(data.message);
                    document.getElementById("rfqForm").reset();
                    document.querySelectorAll('.opt-input').forEach(el => el.classList.add('d-none'));
                    document.getElementById('inp_info_container').classList.add('d-none');
                    document.getElementById("sendToAllRadio").checked = true;
                    toggleCarrierSelection();
                    loadStats();
                    loadRequests();
                } else {
                    alert(data.detail || "Sorğu yaradılarkən xəta baş verdi!");
                }
            } catch (err) {
                console.error(err);
                alert("Server xətası!");
            }
        }

        async function loadRequests() {
            const container = document.getElementById("requestsContainer");
            try {
                const cleanId = typeof user.id === 'string' ? user.id.replace(/[^0-9]/g, '') : user.id;
                const res = await fetch(`/requests/customer/${cleanId}`);
                const data = await res.json();
                if (res.ok && data.requests.length > 0) {
                    let html = "";
                    for (let req of data.requests) {
                        html += `
                            <div class="border rounded-4 p-4 mb-3 bg-light">
                                <div class="d-flex justify-content-between align-items-center mb-2">
                                    <h5 class="fw-bold text-dark mb-0">Sorğu #${req.id}: ${req.origin} ➔ ${req.destination}</h5>
                                    <span class="badge ${req.status === 'open' ? 'bg-primary' : 'bg-success'}">${req.status.toUpperCase()}</span>
                                </div>
                                <p class="text-muted small mb-2"><strong>Yük Növü:</strong> ${req.cargo_type} | <strong>Çəki:</strong> ${req.weight_kg} kq | <strong>Yükləmə Tarixi:</strong> ${req.deadline}</p>
                                <div id="quotes-for-${req.id}" class="mt-3">
                                    <button class="btn btn-sm btn-outline-primary" onclick="loadQuotes(${req.id})">
                                        <i class="bi bi-eye"></i> Təklifləri Göstər
                                    </button>
                                </div>
                            </div>
                        `;
                    }
                    container.innerHTML = html;
                } else {
                    container.innerHTML = `<p class="text-muted small">Hələ heç bir sorğu yaratmamısınız.</p>`;
                }
            } catch (err) {
                container.innerHTML = `<p class="text-danger small">Sorğuları yükləmək mümkün olmadı.</p>`;
            }
        }

        window.currentQuotesCache = {};

        async function loadQuotes(requestId) {
            const quotesDiv = document.getElementById(`quotes-for-${requestId}`);
            quotesDiv.innerHTML = `<span class="spinner-border spinner-border-sm text-primary"></span> Təkliflər yüklənir...`;
            
            try {
                const res = await fetch(`/quotes/request/${requestId}`);
                const data = await res.json();
                if (res.ok && data.quotes.length > 0) {
                    window.currentQuotesCache[requestId] = data.quotes;
                    
                    let html = `<div class="table-responsive mt-2"><table class="table table-bordered bg-white align-middle small">
                        <thead>
                            <tr class="table-light">
                                <th>Daşıyıcı Şirkət</th>
                                <th>Qiymət</th>
                                <th>Tranzit Müddəti</th>
                                <th>Əlavə Detallar / Qeydlər</th>
                                <th>Əməliyyat</th>
                            </tr>
                        </thead>
                        <tbody>`;
                    
                    for (let q of data.quotes) {
                        let extraStr = "";
                        if (q.extra_details) {
                            for (let [k, v] of Object.entries(q.extra_details)) {
                                extraStr += `<div><strong>${k}:</strong> ${v}</div>`;
                            }
                        }

                        html += `
                            <tr class="${q.is_winner ? 'table-success' : ''}">
                                <td class="fw-semibold">${q.carrier_company}</td>
                                <td class="text-success fw-bold">${q.price} ${q.currency || 'AZN'}</td>
                                <td>${q.transit_time_days} gün</td>
                                <td>
                                    <div class="truncate-text">${extraStr || '<span class="text-muted">Yoxdur</span>'}</div>
                                    <button class="btn btn-link btn-sm p-0 text-primary mt-1" onclick="openQuoteModal(${requestId}, ${q.id})">
                                        <i class="bi bi-arrows-fullscreen"></i> Detallı Bax
                                    </button>
                                </td>
                                <td>
                                    ${q.is_winner ? 
                                        '<span class="badge bg-success">Qalib</span>' : 
                                        `<button class="btn btn-sm btn-success" onclick="selectWinner(${requestId}, ${q.id})">Qalib Seç</button>`
                                    }
                                </td>
                            </tr>
                        `;
                    }
                    html += `</tbody></table></div>`;
                    quotesDiv.innerHTML = html;
                } else {
                    quotesDiv.innerHTML = `<p class="text-muted small mt-2">Hələ bu sorğuya heç bir daşıyıcı təklif göndərməyib.</p>`;
                }
            } catch (err) {
                quotesDiv.innerHTML = `<p class="text-danger small mt-2">Təklifləri yükləmək xətası.</p>`;
            }
        }

        function openQuoteModal(requestId, quoteId) {
            const quotes = window.currentQuotesCache[requestId] || [];
            const q = quotes.find(item => item.id === quoteId);
            if (!q) return;

            let extraHtml = "";
            if (q.extra_details) {
                for (let [k, v] of Object.entries(q.extra_details)) {
                    extraHtml += `<div class="mb-2 p-3 bg-light rounded-3 border"><strong>${k}:</strong> <p class="mb-0 mt-1 text-dark" style="white-space: pre-wrap;">${v}</p></div>`;
                }
            }

            const modalContent = document.getElementById("modalQuoteContent");
            modalContent.innerHTML = `
                <div class="mb-3">
                    <h6 class="text-muted small">Daşıyıcı Şirkət</h6>
                    <h4 class="fw-bold text-primary">${q.carrier_company}</h4>
                </div>
                <div class="row g-3 mb-3">
                    <div class="col-md-6">
                        <div class="p-3 bg-light rounded-3">
                            <span class="text-muted small">Təklif olunan Qiymət</span>
                            <h3 class="text-success fw-bold mb-0">${q.price} ${q.currency || 'AZN'}</h3>
                        </div>
                    </div>
                    <div class="col-md-6">
                        <div class="p-3 bg-light rounded-3">
                            <span class="text-muted small">Tranzit Müddəti</span>
                            <h3 class="text-dark fw-bold mb-0">${q.transit_time_days} gün</h3>
                        </div>
                    </div>
                </div>
                <div>
                    <h6 class="fw-bold text-dark mb-2">Bütün Əlavə Məlumatlar və Cavablar:</h6>
                    ${extraHtml || '<p class="text-muted">Əlavə detal qeyd olunmayıb.</p>'}
                </div>
            `;

            const modal = new bootstrap.Modal(document.getElementById("quoteDetailModal"));
            modal.show();
        }

        async function selectWinner(requestId, quoteId) {
            if (!confirm("Bu təklifi qalib seçmək istədiyinizə əminsiniz?")) return;
            try {
                const res = await fetch("/quotes/select-winner", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({ request_id: requestId, quote_id: quoteId })
                });
                const data = await res.json();
                if (res.ok) {
                    alert("Qalib təklif uğurla təsdiqləndi!");
                    loadRequests();
                    loadStats();
                } else {
                    alert("Xəta baş verdi!");
                }
            } catch (err) {
                alert("Server xətası!");
            }
        }
    </script>
